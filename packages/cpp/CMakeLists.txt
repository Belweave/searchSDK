cmake_minimum_required(VERSION 3.20)
project(searchsdk_cpp VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(SEARCHSDK_CPP_USE_INSTALLED_C "Find installed C library instead of composing sibling source" OFF)
option(SEARCHSDK_CPP_CURL "Build optional libcurl transport facade" OFF)
option(SEARCHSDK_CPP_BUILD_TESTS "Build deterministic fixture and ownership tests" ON)
option(SEARCHSDK_CPP_SANITIZERS "Instrument C and C++ with ASan/UBSan" OFF)
set(SEARCHSDK_CPP_C_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../c" CACHE PATH "Sibling C package source")
set(SEARCHSDK_CPP_FIXTURES "${CMAKE_CURRENT_SOURCE_DIR}/../../sdk-contract/fixtures" CACHE PATH "Shared deterministic fixture directory")

if(NOT TARGET searchsdk::searchsdk)
    if(SEARCHSDK_CPP_USE_INSTALLED_C)
        find_package(searchsdk 0.1 CONFIG REQUIRED)
    elseif(EXISTS "${SEARCHSDK_CPP_C_SOURCE}/CMakeLists.txt")
        # Scope C defaults locally; do not overwrite an enclosing project's cache settings.
        function(searchsdk_cpp_add_c)
            set(SEARCHSDK_TESTS OFF)
            set(SEARCHSDK_CURL ${SEARCHSDK_CPP_CURL})
            set(SEARCHSDK_SANITIZERS ${SEARCHSDK_CPP_SANITIZERS})
            add_subdirectory("${SEARCHSDK_CPP_C_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/c")
        endfunction()
        searchsdk_cpp_add_c()
    else()
        message(FATAL_ERROR "C library missing. Supply SEARCHSDK_CPP_C_SOURCE or SEARCHSDK_CPP_USE_INSTALLED_C=ON.")
    endif()
endif()

add_library(searchsdk_cpp src/searchsdk.cpp)
add_library(searchsdk::cpp ALIAS searchsdk_cpp)
set_target_properties(searchsdk_cpp PROPERTIES EXPORT_NAME cpp VERSION 0.1.0 SOVERSION 0 POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(searchsdk_cpp PUBLIC cxx_std_17)
target_include_directories(searchsdk_cpp PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(searchsdk_cpp PUBLIC searchsdk::searchsdk)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(searchsdk_cpp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SEARCHSDK_CPP_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "ASan/UBSan requires Clang or GCC")
    endif()
    target_compile_options(searchsdk_cpp PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(searchsdk_cpp PUBLIC -fsanitize=address,undefined)
endif()

set(searchsdk_cpp_install_targets searchsdk_cpp)
if(SEARCHSDK_CPP_CURL)
    if(NOT TARGET searchsdk::curl)
        message(FATAL_ERROR "Enable SEARCHSDK_CURL when configuring the C library to build searchsdk::cpp_curl")
    endif()
    find_package(CURL 7.85 REQUIRED)
    add_library(searchsdk_cpp_curl src/curl.cpp)
    add_library(searchsdk::cpp_curl ALIAS searchsdk_cpp_curl)
    set_target_properties(searchsdk_cpp_curl PROPERTIES EXPORT_NAME cpp_curl POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
    target_link_libraries(searchsdk_cpp_curl PUBLIC searchsdk_cpp searchsdk::curl PRIVATE CURL::libcurl)
    list(APPEND searchsdk_cpp_install_targets searchsdk_cpp_curl)
endif()

install(TARGETS ${searchsdk_cpp_install_targets} EXPORT searchsdkCppTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE README.md native-manifest.json package.json DESTINATION ${CMAKE_INSTALL_DATADIR}/searchsdk-cpp)
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/searchsdk-cpp/examples)
install(EXPORT searchsdkCppTargets NAMESPACE searchsdk:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/searchsdk_cpp)
configure_package_config_file(cmake/searchsdk_cppConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/searchsdk_cppConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/searchsdk_cpp)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/searchsdk_cppConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/searchsdk_cppConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/searchsdk_cppConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/searchsdk_cpp)
file(RELATIVE_PATH SEARCHSDK_CPP_PC_PREFIX "${CMAKE_INSTALL_FULL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
configure_file(cmake/searchsdk-cpp.pc.in searchsdk-cpp.pc @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/searchsdk-cpp.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)

if(SEARCHSDK_CPP_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(searchsdk_cpp_tests tests/test.cpp)
    target_link_libraries(searchsdk_cpp_tests PRIVATE searchsdk_cpp Threads::Threads)
    target_compile_definitions(searchsdk_cpp_tests PRIVATE SEARCHSDK_FIXTURES="${SEARCHSDK_CPP_FIXTURES}")
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(searchsdk_cpp_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME cpp_contract COMMAND searchsdk_cpp_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/providers-a/operations.json")
        add_test(NAME cpp_generated COMMAND ${Python3_EXECUTABLE} -B "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate.py" --check)
    endif()
    if(SEARCHSDK_CPP_CURL AND Python3_FOUND)
        add_executable(searchsdk_cpp_loopback tests/loopback.cpp)
        target_link_libraries(searchsdk_cpp_loopback PRIVATE searchsdk_cpp_curl Threads::Threads)
        add_test(NAME cpp_https_loopback COMMAND ${Python3_EXECUTABLE} -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/loopback.py" $<TARGET_FILE:searchsdk_cpp_loopback> "${SEARCHSDK_CPP_FIXTURES}/search-success.json")
        set_tests_properties(cpp_https_loopback PROPERTIES TIMEOUT 30)
    endif()
endif()

set(CPACK_PACKAGE_NAME searchsdk-cpp)
set(CPACK_PACKAGE_VERSION "0.1.0-alpha.1")
set(CPACK_PACKAGE_VENDOR Belweave)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "C++17 RAII facade over the searchSDK C ABI")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_GENERATOR TGZ)
include(CPack)
