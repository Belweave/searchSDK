cmake_minimum_required(VERSION 3.20)
project(searchsdk_cpp_consumer LANGUAGES CXX)
find_package(searchsdk_cpp 0.1 CONFIG REQUIRED)
add_executable(consumer consumer.cpp)
if(TARGET searchsdk::cpp_curl)
    target_link_libraries(consumer PRIVATE searchsdk::cpp_curl)
    target_compile_definitions(consumer PRIVATE SEARCHSDK_EXAMPLE_HAS_CURL=1)
else()
    target_link_libraries(consumer PRIVATE searchsdk::cpp)
endif()
enable_testing()
add_test(NAME installed_consumer COMMAND consumer)
