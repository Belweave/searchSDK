cmake_minimum_required(VERSION 3.20)
project(searchsdk_consumer C)
find_package(searchsdk 0.1 CONFIG REQUIRED)
add_executable(searchsdk_consumer consumer.c)
target_compile_features(searchsdk_consumer PRIVATE c_std_11)
if(TARGET searchsdk::curl)
 target_compile_definitions(searchsdk_consumer PRIVATE SSK_HAS_CURL=1)
 target_link_libraries(searchsdk_consumer PRIVATE searchsdk::curl)
else()
 target_link_libraries(searchsdk_consumer PRIVATE searchsdk::searchsdk)
endif()
