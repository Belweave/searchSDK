cmake_minimum_required(VERSION 3.20)
project(searchsdk VERSION 0.1.0 LANGUAGES C)
if(NOT UNIX)
 message(FATAL_ERROR "searchSDK C alpha requires a POSIX platform (macOS or Linux)")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(SEARCHSDK_CURL "Build optional system libcurl transport" ON)
option(SEARCHSDK_TESTS "Build deterministic tests" ON)
option(SEARCHSDK_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(Threads REQUIRED)
add_library(searchsdk src/json.c src/validation.c src/transport.c src/providers.c src/client.c src/native.c src/operations.c)
add_library(searchsdk::searchsdk ALIAS searchsdk)
set_target_properties(searchsdk PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO VERSION 0.1.0 SOVERSION 0 POSITION_INDEPENDENT_CODE ON)
target_include_directories(searchsdk PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(searchsdk PRIVATE Threads::Threads)
if(UNIX AND NOT APPLE)
 target_link_libraries(searchsdk PRIVATE m)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(searchsdk PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(SEARCHSDK_SANITIZERS)
 if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  message(FATAL_ERROR "Sanitizers require Clang or GCC")
 endif()
 target_compile_options(searchsdk PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(searchsdk PUBLIC -fsanitize=address,undefined)
endif()
set(SEARCHSDK_INSTALL_TARGETS searchsdk)
if(SEARCHSDK_CURL)
 find_package(CURL 7.85 REQUIRED)
 add_library(searchsdk_curl src/curl.c)
 add_library(searchsdk::curl ALIAS searchsdk_curl)
 set_target_properties(searchsdk_curl PROPERTIES EXPORT_NAME curl C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON)
 target_link_libraries(searchsdk_curl PUBLIC searchsdk CURL::libcurl)
 list(APPEND SEARCHSDK_INSTALL_TARGETS searchsdk_curl)
endif()
install(TARGETS ${SEARCHSDK_INSTALL_TARGETS} EXPORT searchsdkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE README.md operations.json sources.json DESTINATION ${CMAKE_INSTALL_DATADIR}/searchsdk)
install(EXPORT searchsdkTargets NAMESPACE searchsdk:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/searchsdk)
configure_package_config_file(cmake/searchsdkConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/searchsdkConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/searchsdk)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/searchsdkConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/searchsdkConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/searchsdkConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/searchsdk)
file(RELATIVE_PATH SEARCHSDK_PC_PREFIX "${CMAKE_INSTALL_FULL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
configure_file(cmake/searchsdk.pc.in searchsdk.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/searchsdk.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
if(SEARCHSDK_CURL)
 configure_file(cmake/searchsdk-curl.pc.in searchsdk-curl.pc @ONLY)
 install(FILES ${CMAKE_CURRENT_BINARY_DIR}/searchsdk-curl.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
endif()
if(SEARCHSDK_TESTS)
 enable_testing()
 add_executable(searchsdk_tests tests/test.c)
 target_include_directories(searchsdk_tests PRIVATE src)
 target_link_libraries(searchsdk_tests PRIVATE searchsdk Threads::Threads)
 target_compile_definitions(searchsdk_tests PRIVATE SSK_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../sdk-contract/fixtures")
 add_test(NAME core COMMAND searchsdk_tests)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/providers-a/operations.json")
  add_test(NAME generated COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate.py --check)
 endif()
 if(SEARCHSDK_CURL AND Python3_FOUND)
  add_executable(searchsdk_loopback tests/loopback.c)
  target_link_libraries(searchsdk_loopback PRIVATE searchsdk_curl)
  add_test(NAME https_loopback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/loopback.py $<TARGET_FILE:searchsdk_loopback> ${CMAKE_CURRENT_SOURCE_DIR}/../../sdk-contract/fixtures/search-success.json)
 endif()
endif()
set(CPACK_PACKAGE_NAME searchsdk-c)
set(CPACK_PACKAGE_VERSION "0.1.0-alpha.1")
set(CPACK_PACKAGE_VENDOR Belweave)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "C11 searchSDK core and optional system-libcurl HTTPS transport")
set(CPACK_GENERATOR TGZ)
include(CPack)
